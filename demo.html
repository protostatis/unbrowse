# Install the binary
pip install pyunbrowser  # or: cargo install unbrowser

# Add to Claude Desktop (4 lines of JSON)
{
  "mcpServers": {
    "unbrowser": {
      "command": "unbrowser",
      "args": ["--mcp"]
    }
  }
}

# Or use from Python directly
from unbrowser import Client
with Client() as ub:
  r = ub.navigate("https://news.ycombinator.com")
  print(r.blockmap.title)
